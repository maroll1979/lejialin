// @version=2
// ===================================================================
//  成交量分布 VP-24 · Volume Profile（24 档）  —— 每日 08:00 重置版（严格按需求）
// -------------------------------------------------------------------
//  1. 以上海时间每日 08:00 为新交易日起点，重置当日高低点与逐档成交量
//  2. 把当日价格区间均分为 24 档，按重叠比例摊分成交量
//  3. 中间 70% 成交量所在区间 = 价值区；上沿 VAH，下沿 VAL，最密档 POC
//  4. 收盘 > VAH → 做空区域；收盘 < VAL → 做多区域
//  
//  【必读】本版依赖时间函数 hour() 与自引用 x[1]：
//    · 若报错，请先跑通「滚动窗口版」确认基础函数可用，再回来调这里
//    · 若平台的 hour 是 UTC，把 8 改成 0（北京时间 08:00 = UTC 00:00）
//    · 盘中档位边界随当日高低点扩张，早期 K 线的归属会略有失真，
//      越接近次日 08:00 越准确。若需要全天稳定的参考位，请用滚动窗口版。
// ===================================================================

// ---------- 参数 ----------
VA_PCT = 70   // 价值区包含的成交量百分比，标准值 70

// ---------- 新交易日判定（上海时间 08:00） ----------
// 常见写法三选一，按平台实际支持情况启用：
//   1) isNewDay = hour == 8 && hour[1] != 8        // hour 为北京时间
//   2) isNewDay = hour == 0 && hour[1] != 23       // hour 为 UTC
//   3) isNewDay = day != day[1]                    // 有 day 函数时按自然日切
isNewDay = hour == 8 && hour[1] != 8

// ---------- 当日已走完的 K 线根数 ----------
// 刚重置时只有 1 根 K 线，价格区间极窄，此时算出的 VAH/VAL 没有参考价值，
// 必须设一个最小根数门槛，否则会在每个 08:00 附近刷出一堆假信号。

// ---------- 当日高低点（重置式累积） ----------

lo = dayLow
rng = hi - lo
step = rng > 0 ? rng / 24 : 0

// ---------- 24 档价格（取每档中点） ----------

ov10 = min(high, lo + 11 * step) - max(low, lo + 10 * step)

// ---------- 当日逐档累积（08:00 重置） ----------
